        val list = mutableListOf<String>()

        for (i in 1..n) {
            val num = when {
                (i % 3 == 0 && i % 5 == 0) -> "FizzBuzz"
                (i % 3 == 0) -> "Fizz"
                (i % 5 == 0) -> "Buzz"
                else -> i.toString()
            }

            list.add(num)
        }

        return list
