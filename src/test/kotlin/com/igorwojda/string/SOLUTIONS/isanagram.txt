        val string1 = str1.lowercase().filter { it.isLetter() }
        val string2 = str2.lowercase().filter { it.isLetter() }

        if (string1.length != string2.length)
            return false

        for (i in string1.indices) {
            val strChar = string1[i]

            if (
                string1.count { it == strChar } !=
                string2.count { it == strChar }
            )
                return false
        }

        return true
